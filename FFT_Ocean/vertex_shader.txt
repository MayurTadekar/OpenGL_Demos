#version 450 core                         

in vec4 vPosition;                        
in vec4 vColor;                           
uniform mat4 u_mMatrix;                 
uniform mat4 u_vMatrix;                 
uniform mat4 u_pMatrix;                 
out vec4 out_Color;

void main(void)                           
{                                         
        gl_Position = u_pMatrix * u_vMatrix * u_mMatrix * vPosition;    
        out_Color = vColor;                       
};
